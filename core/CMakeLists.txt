cmake_minimum_required(VERSION 3.21)
project(grandiniu_ld VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LD_SANITIZE "Enable address/undefined sanitizers for native tests" OFF)
if(LD_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(LD_DEPS "" CACHE PATH "Verified dependencies (download with tools/fetch_core_deps.py)")
if(NOT EXISTS "${LD_DEPS}/eigen-5.0.0/Eigen/Dense" OR NOT EXISTS "${LD_DEPS}/json.hpp")
  message(FATAL_ERROR "Run python tools/fetch_core_deps.py DEST and set -DLD_DEPS=DEST")
endif()
file(SHA256 "${LD_DEPS}/json.hpp" JSON_SHA)
file(SHA256 "${LD_DEPS}/eigen-5.0.0.tar.gz" EIGEN_SHA)
if(NOT JSON_SHA STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63" OR NOT EIGEN_SHA STREQUAL "315c881e19e17542a7d428c5aa37d113c89b9500d350c433797b730cd449c056")
  message(FATAL_ERROR "Dependency checksum mismatch")
endif()
add_library(ldcore SHARED src/mna.cpp src/model.cpp src/sources.cpp src/grader.cpp src/batch.cpp src/report.cpp)
file(ARCHIVE_EXTRACT INPUT "${LD_DEPS}/eigen-5.0.0.tar.gz" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/verified-deps")
target_include_directories(ldcore PUBLIC include)
target_include_directories(ldcore SYSTEM PRIVATE "${LD_DEPS}" "${CMAKE_CURRENT_BINARY_DIR}/verified-deps/eigen-5.0.0")
target_compile_definitions(ldcore PRIVATE EIGEN_MPL2_ONLY)
set_target_properties(ldcore PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(ldcheck src/main.cpp)
target_link_libraries(ldcheck PRIVATE ldcore)
target_include_directories(ldcheck SYSTEM PRIVATE "${LD_DEPS}")
add_executable(mokytojas src/mokytojas.cpp)
target_link_libraries(mokytojas PRIVATE ldcore)
target_include_directories(mokytojas SYSTEM PRIVATE "${LD_DEPS}")
if(WIN32)
  target_link_libraries(mokytojas PRIVATE ws2_32)
  target_link_libraries(ldcore PRIVATE shell32)
endif()
foreach(t ldcore ldcheck mokytojas)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive- /EHsc)
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(APPLE)
  set_target_properties(ldcheck mokytojas PROPERTIES INSTALL_RPATH "@loader_path")
else()
  set_target_properties(ldcheck mokytojas PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS ldcore ldcheck mokytojas RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
enable_testing()
add_executable(layoutcheck tests/layout_check.cpp)
if(MSVC)
  target_compile_options(layoutcheck PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(layoutcheck PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME layout_geometry COMMAND layoutcheck --self-test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME grading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_grading.py" "$<TARGET_FILE:ldcheck>")
add_test(NAME student_delivery COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_delivery.py" "$<TARGET_FILE:ldcore>" "$<TARGET_FILE:ldcheck>")
if(NOT LD_SANITIZE)
  add_test(NAME native_abi COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_native.py" "$<TARGET_FILE:ldcore>")
endif()
# Instrukcijų <-> registrų konsistencija (tas pats Python interpretatorius kaip kiti testai).
add_test(NAME instruction_registry COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/check_instruction_registry.py" --root "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME mokytojas COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mokytojas.py" "$<TARGET_FILE:mokytojas>")
add_test(NAME mokytojas_ui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mokytojas_ui.py" "$<TARGET_FILE:mokytojas>")

add_test(NAME ld5_grading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ld5_grading.py" "$<TARGET_FILE:ldcheck>")
add_test(NAME ld6_sources COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ld6_grading.py" "$<TARGET_FILE:ldcheck>" "$<TARGET_FILE:ldcore>")
add_test(NAME ld7_grading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ld7_grading.py" "$<TARGET_FILE:ldcheck>" "$<TARGET_FILE:ldcore>")

add_test(NAME ld10_grading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ld10_grading.py" "$<TARGET_FILE:ldcheck>" "$<TARGET_FILE:ldcore>")
add_test(NAME ld11_grading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ld11_grading.py" "$<TARGET_FILE:ldcheck>" "$<TARGET_FILE:ldcore>")
add_test(NAME ld12_grading COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ld12_grading.py" "$<TARGET_FILE:ldcheck>" "$<TARGET_FILE:ldcore>")
